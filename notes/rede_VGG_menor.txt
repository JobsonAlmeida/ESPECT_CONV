class SpaceFrequencyCNN(nn.Module):

    def __init__(self):

        super().__init__()

        # Entrada:
        # (batch, channels, frequency, row, column)
        #
        # Exemplo:
        # (8, 9, 65, 21, 21)

        # -------------------------------------------------
        # Bloco convolucional 1
        # -------------------------------------------------

        self.conv1 = nn.Conv3d(
            in_channels=9,
            out_channels=16,
            kernel_size=3,
            padding=1
        )
        # Saída:
        # (batch, 16, 65, 21, 21)

        self.bn1 = nn.BatchNorm3d(
            num_features=16
        )
        # Saída:
        # (batch, 16, 65, 21, 21)

        self.relu1 = nn.ReLU()

        self.pool1 = nn.MaxPool3d(
            kernel_size=(1, 2, 2)
        )
        # Saída:
        # (batch, 16, 65, 10, 10)


        # -------------------------------------------------
        # Bloco convolucional 2
        # -------------------------------------------------

        self.conv2 = nn.Conv3d(
            in_channels=16,
            out_channels=32,
            kernel_size=3,
            padding=1
        )
        # Saída:
        # (batch, 32, 65, 10, 10)

        self.bn2 = nn.BatchNorm3d(
            num_features=32
        )
        # Saída:
        # (batch, 32, 65, 10, 10)

        self.relu2 = nn.ReLU()

        self.pool2 = nn.MaxPool3d(
            kernel_size=(1, 2, 2)
        )
        # Saída:
        # (batch, 32, 65, 5, 5)


        # -------------------------------------------------
        # Global Average Pooling
        # -------------------------------------------------

        self.global_pool = nn.AdaptiveAvgPool3d(
            output_size=(1, 1, 1)
        )
        # Saída:
        # (batch, 32, 1, 1, 1)


        # -------------------------------------------------
        # Flatten
        # -------------------------------------------------

        self.flatten = nn.Flatten()
        # Saída:
        # (batch, 32)


        # -------------------------------------------------
        # Dropout
        # -------------------------------------------------

        self.dropout = nn.Dropout(
            p=0.5
        )
        # Saída:
        # (batch, 32)


        # -------------------------------------------------
        # Classificador
        # -------------------------------------------------

        self.classifier = nn.Linear(
            in_features=32,
            out_features=4
        )
        # Saída:
        # (batch, 4)


    def forward(self, x):

        # Bloco 1
        x = self.conv1(x)
        x = self.bn1(x)
        x = self.relu1(x)
        x = self.pool1(x)

        # Bloco 2
        x = self.conv2(x)
        x = self.bn2(x)
        x = self.relu2(x)
        x = self.pool2(x)

        # Global Average Pooling
        x = self.global_pool(x)

        # Flatten
        x = self.flatten(x)

        # Dropout
        x = self.dropout(x)

        # Classificador
        x = self.classifier(x)

        return x
